async function loadRCA() { const container = document.getElementById('rcaContainer'); container.innerHTML = 'Loading...'; try { const resp = await fetch(`${API_BASE}/rca/keyword?limit=50`); const data = await resp.json(); // Show the server's own message if the response isn't OK if (!resp.ok) { container.innerHTML = `${data.detail || 'Unknown RCA error'}`; return; } // Success – render the RCA content (unchanged) const keywords = data.keyword_counts || {}; const records = data.matching_records || []; const totalMatches = data.total_matches; const totalRecords = data.total_records; const impact = totalRecords ? ((totalMatches / totalRecords) * 100).toFixed(1) : 0; container.innerHTML = `

🚨 Affected Rows

${totalMatches}

📋 Total Rows

${totalRecords}

📉 Impact Rate

${impact}%

${Object.keys(keywords).length ? `

Top Negative Keywords

${Object.entries(keywords).map(([kw, count]) => `
${kw}
${count}
`).join('')}
` : '

No negative keywords detected.

'} ${records.length ? `

Matching Feedback Rows (${records.length})

${records.map(r => ``).join('')}
ScoreFeedbackMatched Keywords
${r.score || ''}${r.feedback}${r.matched_keywords}
` : '

No matching records found.

'} `; } catch (err) { container.innerHTML = '❌ Failed to load RCA'; } }
